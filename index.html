To connect this form to your Google Sheet, follow these steps:

  1. Create a new Google Sheet.
  2. Create headers in the first row exactly as follows:
    Date | C-Note No | Party Name | Destination | Weight (Kg) | Amount
  3. Go to Extensions > Apps Script.
  4. Paste the following code:
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var rowData = [e.parameter['Date'] || '', e.parameter['C-Note No'] || '', e.parameter['Party Name'] || '', e.parameter['Destination'] || '', e.parameter['Weight (Kg)'] || '', e.parameter['Amount'] || ''];
    sheet.appendRow(rowData);
    return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT);
  } catch (error) { return ContentService.createTextOutput("Error").setMimeType(ContentService.MimeType.TEXT); }
}

function doGet(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = sheet.getDataRange().getDisplayValues(); // getDisplayValues safely handles dates and numbers
    var action = e.parameter.action;
    var query = e.parameter.search ? e.parameter.search.toLowerCase() : "";
    var results = [];
    if (data.length > 1) {
      var headers = data[0];
      for (var i = 1; i < data.length; i++) {
        var row = data[i];
        if(!row[0] && !row[1] && !row[2]) continue; // Skip empty rows
        
        var isToday = false;
        if (action === 'today') {
          var today = new Date();
          var rowDate = new Date(row[0]);
          if (!isNaN(rowDate) && rowDate.getDate() == today.getDate() && rowDate.getMonth() == today.getMonth() && rowDate.getFullYear() == today.getFullYear()) {
            isToday = true;
          }
        }
        
        var matchesSearch = false;
        if (query && (row[2].toString().toLowerCase().indexOf(query) !== -1 || row[1].toString().toLowerCase().indexOf(query) !== -1)) {
          matchesSearch = true;
        }
        
        if (isToday || matchesSearch) {
          var obj = {};
          for (var j = 0; j < headers.length; j++) { obj[headers[j]] = row[j]; }
          results.push(obj);
        }
      }
    }
    return ContentService.createTextOutput(JSON.stringify(results)).setMimeType(ContentService.MimeType.JSON);
  } catch (error) { return ContentService.createTextOutput(JSON.stringify([{error: error.toString()}])).setMimeType(ContentService.MimeType.JSON); }
}
