To connect this form to your Google Sheet, follow these steps:

  1. Create a new Google Sheet.
  2. Create headers in the first row exactly as follows:
    Date | C-Note No | Party Name | Destination | Weight (Kg) | Amount
  3. Go to Extensions > Apps Script.
  4. Paste the following code (UPDATED for Edit, Delete, Custom Dates & Party Autocomplete):
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var action = e.parameter.action;
    
    if (action === 'delete') {
      sheet.deleteRow(parseInt(e.parameter.rowIndex));
    } else if (action === 'edit') {
      var rowData = [e.parameter['Date'] || '', e.parameter['C-Note No'] || '', e.parameter['Party Name'] || '', e.parameter['Destination'] || '', e.parameter['Weight (Kg)'] || '', e.parameter['Amount'] || ''];
      sheet.getRange(parseInt(e.parameter.rowIndex), 1, 1, 6).setValues([rowData]);
    } else {
      var rowData = [e.parameter['Date'] || '', e.parameter['C-Note No'] || '', e.parameter['Party Name'] || '', e.parameter['Destination'] || '', e.parameter['Weight (Kg)'] || '', e.parameter['Amount'] || ''];
      sheet.appendRow(rowData);
    }
    return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT);
  } catch (error) { return ContentService.createTextOutput("Error").setMimeType(ContentService.MimeType.TEXT); }
}

function doGet(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var action = e.parameter.action;
    
    // Autocomplete for Party Names
    if (action === 'parties') {
      var data = sheet.getRange("C2:C").getDisplayValues();
      var uniqueParties = [];
      var partyMap = {};
      for (var i = 0; i < data.length; i++) {
        var party = data[i][0].trim();
        if (party && !partyMap[party]) {
          partyMap[party] = true;
          uniqueParties.push(party);
        }
      }
      return ContentService.createTextOutput(JSON.stringify(uniqueParties)).setMimeType(ContentService.MimeType.JSON);
    }

    var data = sheet.getDataRange().getDisplayValues(); 
    var dateQuery = e.parameter.date;
    var query = e.parameter.search ? e.parameter.search.toLowerCase() : "";
    var results = [];
    if (data.length > 1) {
      for (var i = 1; i < data.length; i++) {
        var row = data[i];
        if(!row[0] && !row[1] && !row[2]) continue; // Skip empty rows
        
        var matchesDate = false;
        if (dateQuery) {
          var qDate = new Date(dateQuery);
          var rowDate = new Date(row[0]);
          if (!isNaN(rowDate) && !isNaN(qDate) && rowDate.getDate() == qDate.getDate() && rowDate.getMonth() == qDate.getMonth() && rowDate.getFullYear() == qDate.getFullYear()) {
            matchesDate = true;
          }
        }
        
        var matchesSearch = false;
        if (query && (row[2].toString().toLowerCase().indexOf(query) !== -1 || row[1].toString().toLowerCase().indexOf(query) !== -1)) {
          matchesSearch = true;
        }
        
        if (matchesDate || matchesSearch) {
          results.push({
            'rowIndex': i + 1, // Capture Row Index for editing/deleting
            'Date': row[0],
            'C-Note No': row[1],
            'Party Name': row[2],
            'Destination': row[3],
            'Weight (Kg)': row[4],
            'Amount': row[5]
          });
        }
      }
    }
    return ContentService.createTextOutput(JSON.stringify(results)).setMimeType(ContentService.MimeType.JSON);
  } catch (error) { return ContentService.createTextOutput(JSON.stringify([{error: error.toString()}])).setMimeType(ContentService.MimeType.JSON); }
}
